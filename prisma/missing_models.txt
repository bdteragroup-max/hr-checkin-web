
model clothing_items {
  id             Int       @id @default(autoincrement())
  name           String    @db.VarChar(200)
  description    String?   @db.Text
  image_url      String?   @db.Text
  is_active      Boolean   @default(true)
  created_at     DateTime  @default(now()) @db.Timestamptz(6)
  updated_at     DateTime  @default(now()) @db.Timestamptz(6)

  variants       clothing_variants[]
}

model clothing_variants {
  id             Int       @id @default(autoincrement())
  item_id        Int
  size           String    @db.VarChar(50)
  stock_quantity Int       @default(0)

  item           clothing_items @relation(fields: [item_id], references: [id], onDelete: Cascade)
  requests       clothing_requests[]

  @@unique([item_id, size])
}

model clothing_requests {
  id               Int       @id @default(autoincrement())
  emp_id           String    @db.VarChar(20)
  variant_id       Int
  quantity         Int       @default(1)
  reason           String?   @db.Text
  status           String    @default("pending") @db.VarChar(50) // pending, approved, rejected, fulfilled
  requested_at     DateTime  @default(now()) @db.Timestamptz(6)
  
  approved_by      String?   @db.VarChar(100) 
  approved_at      DateTime? @db.Timestamptz(6)
  fulfilled_at     DateTime? @db.Timestamptz(6)
  admin_note       String?   @db.Text

  employee         employees          @relation(fields: [emp_id], references: [emp_id], onDelete: Cascade)
  variant          clothing_variants  @relation(fields: [variant_id], references: [id], onDelete: Cascade)

  @@index([emp_id])
  @@index([variant_id])
  @@index([status])
}
